    <script>
        const canvas = document.getElementById('hypnoCanvas');
        const ctx = canvas.getContext('2d');
        let active = false, isIntro = false, time = 0, swing = 0;
        let audioCtx, nodes = [];
        let preValue = '', postValue = '';

        function resize() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
        }
        window.addEventListener('resize', resize);
        resize();

        // --- RENDERER ---
        function drawSpiral(cx, cy) {
            const bands = 45, maxR = Math.max(canvas.width, canvas.height) * 1.5;
            ctx.save();
            ctx.translate(cx, cy);
            ctx.rotate(time * 0.25);
            for (let i = bands; i > 0; i--) {
                const r = (i * (maxR / bands) + (time * 120)) % maxR;
                ctx.beginPath(); ctx.arc(0, 0, Math.abs(r), 0, Math.PI * 2);
                ctx.strokeStyle = i % 2 === 0 ? "#fff" : "#000";
                ctx.lineWidth = (maxR / bands) * 0.9; ctx.stroke();
            }
            ctx.restore();
            time += active ? 0.03 : 0.005;
        }

        function drawAnatomy(cx, cy) {
            const len = 320, amp = Math.PI / 4;
            const angle = Math.sin(swing) * amp;
            if (active && !isIntro) swing += 0.04;

            ctx.save();
            ctx.translate(cx, cy);

            const drawPart = (ox, oy) => {
                const g = ctx.createRadialGradient(ox, oy, 0, ox, oy, 70);
                g.addColorStop(0, "#fce4d6"); g.addColorStop(0.4, "#d08b74"); g.addColorStop(0.9, "#8d5543"); g.addColorStop(1, "#000");
                ctx.fillStyle = g; ctx.beginPath(); ctx.arc(ox, oy, 62, 0, Math.PI * 2); ctx.fill();
            };
            drawPart(-55, 20); drawPart(55, 20);

            ctx.rotate(angle);
            const shaftG = ctx.createLinearGradient(-45, 0, 45, 0);
            shaftG.addColorStop(0, "#8d5543"); shaftG.addColorStop(0.3, "#d08b74"); shaftG.addColorStop(0.5, "#fce4d6"); shaftG.addColorStop(0.7, "#d08b74"); shaftG.addColorStop(1, "#8d5543");
            ctx.fillStyle = shaftG; ctx.beginPath(); ctx.roundRect(-42, -len, 84, len, [35, 35, 0, 0]); ctx.fill();

            // Żyły
            ctx.beginPath(); ctx.moveTo(-15, -60); ctx.bezierCurveTo(-35, -140, 25, -200, 10, -280);
            ctx.strokeStyle = "rgba(100, 50, 180, 0.15)"; ctx.lineWidth = 3; ctx.stroke();

            const glansG = ctx.createRadialGradient(0, -len - 15, 8, 0, -len - 15, 75);
            glansG.addColorStop(0, "#e894a4"); glansG.addColorStop(0.5, "#c46a80"); glansG.addColorStop(1, "#3b1622");
            ctx.fillStyle = glansG; ctx.beginPath(); ctx.ellipse(0, -len - 10, 60, 72, 0, 0, Math.PI * 2); ctx.fill();

            if (active && !isIntro) {
                const p = (Math.sin(time * 15) + 1) * 4;
                ctx.beginPath(); ctx.arc(0, -len - 40, 6 + p, 0, Math.PI * 2);
                ctx.fillStyle = "#fff"; ctx.shadowBlur = 40; ctx.shadowColor = "#a78bfa"; ctx.fill();
            }
            ctx.restore();
        }

        function loop() {
            const cx = canvas.width / 2, cy = canvas.height / 2;
            ctx.fillStyle = "#000"; ctx.fillRect(0, 0, canvas.width, canvas.height);
            drawSpiral(cx, cy);
            const v = ctx.createRadialGradient(cx, cy, 0, cx, cy, Math.max(cx, cy));
            v.addColorStop(0, "rgba(0,0,0,0)"); v.addColorStop(1, "rgba(0,0,0,0.95)");
            ctx.fillStyle = v; ctx.fillRect(0, 0, canvas.width, canvas.height);
            drawAnatomy(cx, cy);
            requestAnimationFrame(loop);
        }
        loop();

        // --- AUDIO ---
        async function setupAudio(play) {
            if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            if (audioCtx.state === 'suspended') await audioCtx.resume();
            if (play) {
                const master = audioCtx.createGain(); master.connect(audioCtx.destination);
                master.gain.setValueAtTime(0, audioCtx.currentTime); master.gain.linearRampToValueAtTime(0.25, audioCtx.currentTime + 5);
                const osc = (f, p, v) => {
                    const o = audioCtx.createOscillator(); const pn = audioCtx.createPanner(); const g = audioCtx.createGain();
                    pn.setPosition(p, 0, 0); o.frequency.value = f; g.gain.value = v;
                    o.connect(pn); pn.connect(g); g.connect(master); o.start(); nodes.push(o, g);
                };
                osc(110, -1, 0.8); osc(114.5, 1, 0.8); osc(55, 0, 0.6); nodes.push(master);
            } else {
                nodes.forEach(n => n.stop ? n.stop() : (n.gain ? n.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 1) : null));
                nodes = [];
            }
        }

        // --- FLOW ---
        function nextScreen(current, next) {
            document.getElementById('screen-' + current).classList.add('hidden');
            document.getElementById('screen-' + next).classList.remove('hidden');
            if (next === 'instructions') runSessionFlow();
        }

        function selectOption(el, type) {
            const container = el.parentElement;
            [...container.children].forEach(c => c.classList.remove('selected'));
            el.classList.add('selected');
            if (type === 'pre') preValue = el.innerText;
            else postValue = el.innerText;
        }

        function validateAndNext(current, next) {
            if (current === 'pre-survey' && !preValue) return alert('Proszę wybrać opcję.');
            if (current === 'post-survey' && !postValue) return alert('Proszę wybrać opcję.');
            nextScreen(current.replace('screen-',''), next);
        }

        async function runSessionFlow() {
            isIntro = true;
            active = true;
            setupAudio(true);
            const fill = document.getElementById('progress-bar-fill');
            const mainT = document.getElementById('ins-main');
            const subT = document.getElementById('ins-sub');
            const steps = [
                { m: "Oddychaj głęboko nosem...", s: "Krok 1: Inicjacja oddechowa" },
                { m: "Zafiksuj wzrok na żołędzi...", s: "Krok 2: Fiksacja wizualna" },
                { m: "Podążaj wzrokiem za ruchem Dicking...", s: "Krok 3: Synchronizacja" },
                { m: "Wszystko inne znika...", s: "Krok 4: Wejście w trans" }
            ];

            let p = 0;
            const interval = setInterval(() => {
                p += (1/30) * 100; fill.style.width = p + '%';
                const idx = Math.floor((p/100) * steps.length);
                if (steps[idx]) { mainT.innerText = steps[idx].m; subT.innerText = steps[idx].s; }
                if (p >= 100) { 
                    clearInterval(interval); 
                    document.getElementById('screen-instructions').classList.add('hidden');
                    isIntro = false;
                    document.getElementById('session-controls').style.opacity = '1';
                    document.getElementById('session-controls').style.pointerEvents = 'auto';
                }
            }, 1000);
        }

        function finishSession() {
            active = false;
            setupAudio(false);
            document.getElementById('session-controls').style.opacity = '0';
            document.getElementById('session-controls').style.pointerEvents = 'none';
            document.getElementById('screen-post-survey').classList.remove('hidden');
        }

        // AI Summary
        async function validateAndNextSummary(current, next) {
            validateAndNext(current, next);
            const summary = document.getElementById('summary-result');
            const apiKey = "";
            try {
                const prompt = `Użytkownik przed sesją hipnozy Dicking czuł się: ${preValue}. Po sesji czuje się: ${postValue}. Napisz krótkie (20 słów) naukowe podsumowanie tej zmiany, wspominając o synchronizacji fal mózgowych Theta.`;
                const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`, {
                    method: 'POST',
                    headers: {'Content-Type': 'application/json'},
                    body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }] })
                });
                const data = await response.json();
                summary.innerText = data.candidates[0].content.parts[0].text;
            } catch (e) {
                summary.innerText = `Twoja transformacja z "${preValue}" do "${postValue}" wskazuje na skuteczne przejście układu nerwowego w stan regeneracji parasympatycznej.`;
            }
        }
    </script>
    <!-- Przeciążenie funkcji dla AI -->
    <script>
        const originalValidate = validateAndNext;
        window.validateAndNext = (current, next) => {
            if (next === 'summary') validateAndNextSummary(current, next);
            else originalValidate(current, next);
        };
    </script>
